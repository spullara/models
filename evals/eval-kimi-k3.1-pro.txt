        if k == 2 and can_diagonal:
            yield "X", r + 1, c + 1
            return

            yield "D", r + 1, c
        if c + 1 < n:
            yield "R", r, c + 1
        if k != 2 and can_diagonal:
            yield "X", r + 1, c + 1

    # dp[r][c][k] is the maximum sum still collectible after leaving (r, c),
    # excluding grid[r][c]. k is the number of moves already made modulo 3.
    dp = [[[None] * 3 for _ in range(n)] for _ in range(n)]
    dp[n - 1][n - 1] = [0, 0, 0]

    # Fill backwards so every destination state is already computed.
    for r in range(n - 1, -1, -1):
        for c in range(n - 1, -1, -1):
            if r == n - 1 and c == n - 1:
                continue

            for k in range(3):
                best = None
                next_k = (k + 1) % 3

                for _, nr, nc in options(r, c, k):
                    value = grid[nr][nc] + dp[nr][nc][next_k]
                    if best is None or value > best:
                        best = value

                dp[r][c][k] = best

    max_sum = grid[0][0] + dp[0][0][0]

    # choosing the first optimal move also gives a deterministic
    # lexicographically smallest optimal path.
