        if not valid_moves:
            return (float('-inf'), "")

        for move_char, ni, nj in valid_moves:
            future_sum, future_path = dp(ni, nj, moves_made + 1)
            if future_sum == float('-inf'):
                continue
            total_sum = grid[ni][nj] + future_sum
            if total_sum > best_sum or (total_sum == best_sum and (best_path == "" or move_char + future_path < best_path)):
                best_sum = total_sum
                best_path = move_char + future_path

        return (best_sum, best_path)

    # Start at (0,0), 0 moves made, count grid[0][0]
    result_sum, result_path = dp(0, 0, 0)
    return (grid[0][0] + result_sum, result_path)
