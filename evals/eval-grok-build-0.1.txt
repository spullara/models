def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    if not grid or not grid[0]:
        return 0, ""
    N = len(grid)
    if N == 1:
        return grid[0][0], ""

    INF = float('-inf')

    path_dp = [[[""] * MAX_K for _ in range(N)] for _ in range(N)]
    dp[0][0][0] = grid[0][0]
    path_dp[0][0][0] = ""
