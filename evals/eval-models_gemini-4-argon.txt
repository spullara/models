    if not grid or not grid[0]:
        return 0, ""

        return grid[0][0], ""

    # where `rem` is (number of moves made so far) % 3
    dp = [[[(-float('inf'), "") for _ in range(3)] for _ in range(N)] for _ in range(N)]
    dp[0][0][0] = (grid[0][0], "")

                    continue

                    moves.append((r + 1, c + 1, 'X'))

                        moves.append((r, c + 1, 'R'))
