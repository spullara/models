    Both the starting and ending cells count toward the sum.
    Time: O(N²). Space: O(N²).

    # when the number of moves already made is phase modulo 3.

    choice = [[[""] * 3 for _ in range(n)] for _ in range(n)]

        for c in range(n - 1, -1, -1):

                moves.append(("R", r, c + 1))
            if r + 1 < n:
                moves.append(("D", r + 1, c))
            diagonal_possible = r + 1 < n and c + 1 < n
            if diagonal_possible:
                moves.append(("X", r + 1, c + 1))

            for phase in range(3):
                # phase == 2 means the next move is a multiple of 3.
                allowed = (
                    [("X", r + 1, c + 1)]
                    if phase == 2 and diagonal_possible
                    else moves
                )
                next_phase = (phase + 1) % 3
                best_sum = None
