def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    """Return the maximum path sum (including both endpoints) and its moves.

    Time: O(N^2). Space: O(N^2).
    Raises ValueError if the grid is empty or not square.
    """
    n = len(grid)
    if n == 0 or any(len(row) != n for row in grid):
        raise ValueError("grid must be a nonempty N×N grid")

    # dp[r][c][k]: maximum remaining sum, including grid[r][c],
    # when the number of moves already taken is congruent to k modulo 3.
    dp = [[[0] * 3 for _ in range(n)] for _ in range(n)]
    choice = [[[""] * 3 for _ in range(n)] for _ in range(n)]

    moves = (("R", 0, 1), ("D", 1, 0), ("X", 1, 1))

    for r in range(n - 1, -1, -1):
        for c in range(n - 1, -1, -1):
            if r == n - 1 and c == n - 1:
                dp[r][c] = [grid[r][c]] * 3
                continue

            for k in range(3):
                next_k = (k + 1) % 3
                force_diagonal = k == 2 and r + 1 < n and c + 1 < n
                allowed = (("X", 1, 1),) if force_diagonal else moves

                best_sum = None
                best_move = ""

                for move, dr, dc in allowed:
                    nr, nc = r + dr, c + dc
                    if nr >= n or nc >= n:
                        continue

                    total = grid[r][c] + dp[nr][nc][next_k]
                    if best_sum is None or total > best_sum:
                        best_sum = total
                        best_move = move

                # Every nonterminal cell has at least one legal move.
                assert best_sum is not None
                dp[r][c][k] = best_sum
                choice[r][c][k] = best_move

    path = []

    while r != n - 1 or c != n - 1:
        move = choice[r][c][k]
        path.append(move)

        if move in ("D", "X"):
            r += 1
        if move in ("R", "X"):
            c += 1
        k = (k + 1) % 3

    return dp[0][0][0], "".join(path)
