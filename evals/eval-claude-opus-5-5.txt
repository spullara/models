This uses dynamic programming. Each state is `(row, col, moves_made % 3)`, because the only history that matters is how many moves have been made modulo 3. Every move increases `row + col`, so cells are processed in increasing order of `row + col` (anti-diagonals). That guarantees every state is final before it is expanded.

**Assumptions:**
- **Sum:** the start and end cells both count toward the sum.
- **Ties:** when two paths have the same sum, the lexicographically smallest path string is returned.
- **Edge cases:** an empty grid returns `(0, "")`, and a 1×1 grid returns `(grid[0][0], "")`.
- **Grid shape:** non-square rectangular grids are also handled.

def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    if not grid or not grid[0]:
        return (0, "")

    rows, cols = len(grid), len(grid[0])

    # best[(i, j, m)] = (max_sum, path) where m = number of moves made so far mod 3
    best = {(0, 0, 0): (grid[0][0], "")}

    def relax(key, cand_sum, cand_path):
        cur = best.get(key)
        if (cur is None or cand_sum > cur[0]
                or (cand_sum == cur[0] and cand_path < cur[1])):
            best[key] = (cand_sum, cand_path)

    # Every move strictly increases i + j, so process cells by anti-diagonal.
    for s in range(rows + cols - 1):
        for i in range(max(0, s - cols + 1), min(rows, s + 1)):
            j = s - i
            for m in range(3):
                state = best.get((i, j, m))
                if state is None:
                    continue
                cur_sum, cur_path = state

                can_r = j + 1 < cols
                can_d = i + 1 < rows
                can_x = can_r and can_d

                # The next move is move number (moves_made + 1).
                # It is a "3rd move" iff (moves_made + 1) % 3 == 0, i.e. m == 2.
                required_diag = (m == 2)
                nm = (m + 1) % 3

                moves = []
                if required_diag and can_x:
                    moves.append(('X', i + 1, j + 1))
                else:
                    if can_r:
                        moves.append(('R', i, j + 1))
                    if can_d:
                        moves.append(('D', i + 1, j))
                    if can_x:
                        moves.append(('X', i + 1, j + 1))

                for mv, ni, nj in moves:
                    relax((ni, nj, nm), cur_sum + grid[ni][nj], cur_path + mv)

    result = None
    for m in range(3):
        st = best.get((rows - 1, cols - 1, m))
        if st is None:
            continue
        if (result is None or st[0] > result[0]
                or (st[0] == result[0] and st[1] < result[1])):
            result = st

    return result if result is not None else (0, "")
