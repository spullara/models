    if not grid or not grid[0]:
        return (0, "")

    n = len(grid)
    neg = float("-inf")

    dp = [[[neg] * 3 for _ in range(n)] for _ in range(n)]

    dp[0][0][0] = grid[0][0]

    for i in range(n):

                    continue

                can_x = i + 1 < n and j + 1 < n
                can_r = j + 1 < n
                can_d = i + 1 < n

                options = []
                if must_x and can_x:
                    options.append(("X", i + 1, j + 1))
                else:
                    if can_r:
                        options.append(("R", i, j + 1))

                        options.append(("D", i + 1, j))

                        options.append(("X", i + 1, j + 1))

                    new_sum = dp[i][j][mod] + grid[ni][nj]

                        parent[ni][nj][nmod] = (i, j, mod, letter)
