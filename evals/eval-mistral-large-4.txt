                    new_sum = curr_sum + grid[i+1][j+1]
                    new_k = (k + 1) % 3
                    new_path = curr_path + "X"
                    # Diagonal is always allowed, and required when must_diagonal
                    if new_sum > dp[i+1][j+1][new_k][0]:
                        dp[i+1][j+1][new_k] = (new_sum, new_path)

    # Find best result at (n-1, m-1) across all k values
    best_sum = -float('inf')
    best_path = ""

    for k in range(3):
        s, p = dp[n-1][m-1][k]
        if s > best_sum:
            best_sum = s
            best_path = p

    return (best_sum, best_path)
