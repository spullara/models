def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    """
    Find the max-sum path from top-left to bottom-right of an N x N grid.

    Moves:
      R = right (c+1), D = down (r+1), X = diagonal down-right (r+1, c+1).
    Rule: every 3rd move (3, 6, 9, ...) must be X if X is possible from the
    current position. If X is not possible on such a move, R or D may be used.

    Returns (max_sum, path_string). The sum includes grid[0][0] and every
    visited cell.
    """
    n = len(grid)
    if n == 0:
        return (0, "")
    NEG = float("-inf")
    end = (n - 1, n - 1)

    def candidates(r, c, k):
        """Yield (move, nr, nc) legal from (r, c) when k moves have been made (mod 3)."""
        diag_ok = r + 1 < n and c + 1 < n
        # The next move is move number (k + 1). It is required to be diagonal
        # when (k + 1) % 3 == 0, i.e. k == 2.
        required = (k == 2)
        if required and diag_ok:
            options = ["X"]
        elif required:
            options = ["R", "D"]
        else:
            options = ["R", "D", "X"]
        for m in options:
            if m == "R" and c + 1 < n:
                yield m, r, c + 1
            elif m == "D" and r + 1 < n:
                yield m, r + 1, c
            elif m == "X" and diag_ok:
                yield m, r + 1, c + 1

    # best[r][c][k] = max sum of cells visited AFTER being at (r, c)
    # with k = (number of moves so far) mod 3, until reaching the end.
    best = [[[NEG] * 3 for _ in range(n)] for _ in range(n)]
    for r in range(n - 1, -1, -1):
        for c in range(n - 1, -1, -1):
            for k in range(3):
                if (r, c) == end:
                    best[r][c][k] = 0
                    continue
                top = NEG
                for _, nr, nc in candidates(r, c, k):
                    sub = best[nr][nc][(k + 1) % 3]
                    if sub > NEG:
                        top = max(top, sub + grid[nr][nc])
                best[r][c][k] = top

    total = grid[0][0] + best[0][0][0]

    # Reconstruct one optimal path.
    path = []
    r, c, k = 0, 0, 0
    while (r, c) != end:
        target = best[r][c][k]
        chosen = None

                break
        if chosen is None:
            # Should not happen for a valid grid; guard against it anyway.
            break
        m, r, c = chosen
        path.append(m)
        k = (k + 1) % 3

    return (int(total), "".join(path))
