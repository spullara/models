        return 0, ""

        else:
            if c + 1 < N:
                candidates.append(("R", 0, 1))
            if r + 1 < N:
                candidates.append(("D", 1, 0))
            if r + 1 < N and c + 1 < N:
                candidates.append(("X", 1, 1))

        if must_diag and not candidates:
            # fallback to R or D
            if c + 1 < N:
                candidates.append(("R", 0, 1))
            if r + 1 < N:
                candidates.append(("D", 1, 0))
