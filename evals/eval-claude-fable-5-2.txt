            can_down = r + 1 < rows

                if cur is None:
                    continue

                nk = (k + 1) % 3

                # It must be diagonal if a diagonal is possible from here.
                if k == 2 and can_diag:
                    moves = (("X", r + 1, c + 1),)

                        moves.append(("X", r + 1, c + 1))

                    if best[nr][nc][nk] is None or val > best[nr][nc][nk]:
