def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    """Return the maximum path sum and its moves, including both endpoints.

    Every third move must be diagonal whenever that move is in bounds.
    Ties may be resolved arbitrarily.

    Time: O(N²). Space: O(N²).
    Raises ValueError if the grid is empty or not square.
    """
    n = len(grid)
    if n == 0 or any(len(row) != n for row in grid):
        raise ValueError("grid must be a nonempty square")

    # dp[r][c][k]: best sum from (r, c) to the destination when
    # the number of moves already taken is congruent to k modulo 3.
    dp = [[[0] * 3 for _ in range(n)] for _ in range(n)]
    choice = [[[""] * 3 for _ in range(n)] for _ in range(n)]

    for r in range(n - 1, -1, -1):
        for c in range(n - 1, -1, -1):
            if r == n - 1 and c == n - 1:
                dp[r][c] = [grid[r][c]] * 3
                continue

            diagonal_possible = r + 1 < n and c + 1 < n

            for k in range(3):
                next_k = (k + 1) % 3

                if k == 2 and diagonal_possible:
                    moves = [("X", r + 1, c + 1)]
                else:
                    moves = []
                    if c + 1 < n:
                        moves.append(("R", r, c + 1))
                    if r + 1 < n:
                        moves.append(("D", r + 1, c))
                    if diagonal_possible:
                        moves.append(("X", r + 1, c + 1))

                move, nr, nc = max(
                    moves,
                    key=lambda candidate: dp[candidate[1]][candidate[2]][next_k],
                )
                dp[r][c][k] = grid[r][c] + dp[nr][nc][next_k]
                choice[r][c][k] = move

    path = []
    r = c = k = 0

    while r != n - 1 or c != n - 1:
        move = choice[r][c][k]
        path.append(move)

        if move in ("D", "X"):
            r += 1
        if move in ("R", "X"):
            c += 1
        k = (k + 1) % 3

    return dp[0][0][0], "".join(path)
