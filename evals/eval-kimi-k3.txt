    """
    Find the maximum-sum path from top-left to bottom-right of an N×N grid.

    Moves: R (right), D (down), X (diagonal down-right).
    Every 3rd move (3, 6, 9, ...) must be X if a diagonal move is possible
    from the current cell; otherwise R or D may be used.

    Returns (max_sum, path_string).
    """
    n = len(grid)
    if n == 0:
        return (0, "")

    NEG = float("-inf")

    # dp[r][c][m] = [best_sum, best_path] for reaching cell (r, c)
    # after making k moves where k % 3 == m.
    dp = [[[[NEG, ""] for _ in range(3)] for _ in range(n)] for _ in range(n)]
    dp[0][0][0] = [grid[0][0], ""]

    for r in range(n):
        for c in range(n):
            for m in range(3):
                cur_sum, cur_path = dp[r][c][m]
                if cur_sum == NEG:
                    continue

                can_r = c + 1 < n
                can_d = r + 1 < n
                can_x = can_r and can_d

                # The next move is move number k+1; it is a multiple of 3
                # exactly when k % 3 == 2.
                next_must_be_diagonal = (m == 2)

                moves = []
                if next_must_be_diagonal:
                    if can_x:
                        moves.append(("X", r + 1, c + 1))
                    else:
                        # Diagonal impossible: R or D are allowed.
                        if can_r:
                            moves.append(("R", r, c + 1))
                        if can_d:
                            moves.append(("D", r + 1, c))
                else:
                    if can_r:
                        moves.append(("R", r, c + 1))
                    if can_d:
                        moves.append(("D", r + 1, c))
                    if can_x:
                        moves.append(("X", r + 1, c + 1))

                nm = (m + 1) % 3
                for ch, nr, nc in moves:
                    cand_sum = cur_sum + grid[nr][nc]
                    cand_path = cur_path + ch
                    if (
                        cand_sum > dp[nr][nc][nm][0]
                        or (cand_sum == dp[nr][nc][nm][0]
                            and cand_path < dp[nr][nc][nm][1])
                    ):
                        dp[nr][nc][nm] = [cand_sum, cand_path]

    best_sum, best_path = max(dp[n - 1][n - 1], key=lambda t: t[0])
    return (best_sum, best_path)
