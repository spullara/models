    dp = [[[NEG_INF for _ in range(N)] for _ in range(N)] for _ in range(max_moves + 2)]
    # parent[k][r][c] stores the character ('R', 'D', 'X') of the move taken to reach (r, c) at step k.
    parent = [[[None for _ in range(N)] for _ in range(N)] for _ in range(max_moves + 2)]

    dp[0][0][0] = grid[0][0]

    for k in range(max_moves + 1):
