def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    n = len(grid)
    if n == 0:
        return (0, "")
    if n == 1:
        return (grid[0][0], "")

    NEG = float("-inf")
    suffix = [[[NEG for _ in range(3)] for _ in range(n)] for _ in range(n)]

    for i in range(n - 1, -1, -1):
        for j in range(n - 1, -1, -1):
            if i == n - 1 and j == n - 1:
                for r in range(3):
                    suffix[i][j][r] = grid[i][j]
                continue

            for r in range(3):
                nr = (r + 1) % 3
                required = nr == 0  # next move is move number divisible by 3
                best = NEG

                if required:
                    if i + 1 < n and j + 1 < n:
                        best = suffix[i + 1][j + 1][nr]
                    else:
                        if j + 1 < n:
                            best = max(best, suffix[i][j + 1][nr])
                        if i + 1 < n:
                            best = max(best, suffix[i + 1][j][nr])
                else:
                    if j + 1 < n:
                        best = max(best, suffix[i][j + 1][nr])
                    if i + 1 < n:
                        best = max(best, suffix[i + 1][j][nr])
                    if i + 1 < n and j + 1 < n:
                        best = max(best, suffix[i + 1][j + 1][nr])

                suffix[i][j][r] = grid[i][j] + best

    max_sum = suffix[0][0][0]
    i = j = r = 0
    path = []
    directions = ((0, 1, "R"), (1, 0, "D"), (1, 1, "X"))

    while not (i == n - 1 and j == n - 1):
        nr = (r + 1) % 3
        required = nr == 0
        diag_possible = i + 1 < n and j + 1 < n
        current = suffix[i][j][r]

        for di, dj, ch in directions:
            ni, nj = i + di, j + dj
            if ni >= n or nj >= n:
                continue

            if required:
                if diag_possible:
                    if ch != "X":
                        continue
                else:
                    if ch == "X":
                        continue

            if grid[i][j] + suffix[ni][nj][nr] == current:
                path.append(ch)
                i, j, r = ni, nj, nr
                break

    return (max_sum, "".join(path))
