def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    if not grid or not grid[0]:
        return (0, "")

        return (grid[0][0], "")

    @lru_cache(None)
    def dfs(r: int, c: int, moves: int) -> tuple[int, str]:

            return (grid[r][c], "")

        candidates = []

            if r + 1 < N and c + 1 < N:
                s, p = dfs(r + 1, c + 1, moves + 1)
                candidates.append((grid[r][c] + s, "X" + p))

                candidates.append((grid[r][c] + s, "R" + p))

                s, p = dfs(r + 1, c, moves + 1)
                candidates.append((grid[r][c] + s, "D" + p))
