    # including grid[r][c], after a number of moves congruent to m mod 3.
    dp = [[[None] * 3 for _ in range(n)] for _ in range(n)]
    choice = [[[None] * 3 for _ in range(n)] for _ in range(n)]

                continue

                    moves = (("X", r + 1, c + 1),)
                else:

                        ("D", r + 1, c),
