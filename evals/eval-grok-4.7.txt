    """Max-sum path from top-left to bottom-right.

    Moves: R (right), D (down), X (diagonal down-right).
    Move numbers 3, 6, 9, ... must be X when that step stays inside the grid;
    otherwise R or D may be used. The returned sum includes every visited cell.
    Ties are broken by the lexicographically smallest path string.
    """

        return (0, "")

    best: list[list[list[tuple[int, str]]]] = [
        [[(0, "") for _ in range(3)] for _ in range(n)] for _ in range(n)
    ]

            for k in range(3):
                if r == n - 1 and c == n - 1:
                    best[r][c][k] = (grid[r][c], "")
                    continue

                can_x = r + 1 < n and c + 1 < n
                moves: list[tuple[str, int, int]] = []

                if must_x and can_x:
                    moves.append(("X", r + 1, c + 1))
                else:
                    if c + 1 < n:
                        moves.append(("R", r, c + 1))
                    if r + 1 < n:
                        moves.append(("D", r + 1, c))
                    if can_x:
                        moves.append(("X", r + 1, c + 1))

                chosen_sum: int | None = None
                chosen_path = ""
                for mv, nr, nc in moves:

                    total = grid[r][c] + sub_sum
                    path = mv + sub_path

                        chosen_sum is None

                        or (total == chosen_sum and path < chosen_path)
                    ):
