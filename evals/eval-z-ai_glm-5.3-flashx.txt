def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    """
    Finds the maximum-sum path from top-left to bottom-right of an N x N grid.

    Moves: R (right), D (down), X (diagonal down-right).
    Constraint: every 3rd move (3rd, 6th, 9th, ...) must be 'X' if a diagonal
    move is possible from the current position; otherwise R or D is allowed.

    Returns (max_sum, path_string).

    Approach: layered dynamic programming. Each state is (row, col, moves % 3),
    tracked layer-by-layer over the exact move count t (0 .. 2N-2), because the
    "must be diagonal" rule depends on the move index modulo 3. Paths of
    different lengths may both end at the bottom-right cell, so the best result
    over all layers is kept.
    """
    n = len(grid)
    if n == 0:
        return (0, "")
    if n == 1:
        return (grid[0][0], "")

    max_moves = 2 * n - 2  # upper bound: all R/D moves

    # dp maps (r, c) -> (best_sum, path_string) for paths using exactly t moves
    dp = {(0, 0): (grid[0][0], "")}
    best_end = None  # (sum, path) at bottom-right cell

    for t in range(max_moves):
        if not dp:
            break
        next_move = t + 1
        must_diag = (next_move % 3 == 0)
        ndp = {}

        for (r, c), (s, p) in dp.items():
            can_r = (c + 1 < n)
            can_d = (r + 1 < n)
            can_x = can_r and can_d

            candidates = []
            if must_diag:
                if can_x:
                    # Diagonal is possible -> it is mandatory
                    candidates.append(('X', r + 1, c + 1))
                else:
                    # Fallback: R or D allowed on a required-diagonal move
                    if can_r:
                        candidates.append(('R', r, c + 1))
                    if can_d:
                        candidates.append(('D', r + 1, c))
            else:
                if can_r:
                    candidates.append(('R', r, c + 1))
                if can_d:
                    candidates.append(('D', r + 1, c))
                if can_x:
                    candidates.append(('X', r + 1, c + 1))

            for mv, nr, nc in candidates:
                ns = s + grid[nr][nc]
                key = (nr, nc)
                cur = ndp.get(key)
                if cur is None or ns > cur[0]:
                    ndp[key] = (ns, p + mv)

        dp = ndp

        end_state = dp.get((n - 1, n - 1))
        if end_state is not None and (best_end is None or end_state[0] > best_end[0]):
            best_end = end_state

    if best_end is None:
        # Should not occur for a non-empty grid, but kept as a safe fallback
        return (grid[0][0], "")
    return best_end
