def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    if not grid or not grid[0]:
        return (0, "")
    
    n = len(grid)
    if n == 1:
        return (grid[0][0], "")
    
    # dp[i][j][k] = (max_sum, path) when arriving at (i,j) after k moves
    # We only need to track k % 3 because the constraint is on every 3rd move
    INF = float('-inf')
    dp = [[[ (INF, "") for _ in range(3)] for _ in range(n)] for _ in range(n)]
    
    # Starting position - 0 moves
    dp[0][0][0] = (grid[0][0], "")
    
    directions = [
        (0, 1, 'R'),   # right
        (1, 0, 'D'),   # down
        (1, 1, 'X')    # diagonal
    ]
    
    for i in range(n):
        for j in range(n):
            for moves in range(3):
                if dp[i][j][moves][0] == INF:
                    continue
                current_sum, path = dp[i][j][moves]
                
                # Next move number will be moves+1
                next_moves = (moves + 1) % 3
                must_diagonal = (next_moves == 0)  # every 3rd move (3,6,9...) → next_moves==0
                
                for di, dj, letter in directions:
                    ni, nj = i + di, j + dj
                    if 0 <= ni < n and 0 <= nj < n:
                        # Check diagonal constraint
                        if must_diagonal and letter != 'X':
                            # If diagonal is possible from current position, we must take it
                            if i + 1 < n and j + 1 < n:
                                continue  # skip R or D when X is possible
                        
                        new_sum = current_sum + grid[ni][nj]
                        new_path = path + letter
                        
                        # Update if better
                        if new_sum > dp[ni][nj][next_moves][0]:
                            dp[ni][nj][next_moves] = (new_sum, new_path)
    
    # Find best path reaching bottom-right regardless of move count mod 3
    best_sum = INF
    best_path = ""
    for k in range(3):

            best_sum = dp[n-1][n-1][k][0]
            best_path = dp[n-1][n-1][k][1]
    
    return (int(best_sum), best_path)
