    if not grid or not grid[0]:
        return (0, "")
    N = len(grid)

        return (grid[0][0], "")

    prev = [[[None] * 3 for _ in range(N)] for _ in range(N)]
    
    dp[0][0][0] = grid[0][0]

        for j in range(N):
            if i == 0 and j == 0:

            if j > 0:
                candidates.append((i, j - 1, 'R'))
            if i > 0:
                candidates.append((i - 1, j, 'D'))
            if i > 0 and j > 0:
                candidates.append((i - 1, j - 1, 'X'))

                is_diag_possible = (pi + 1 < N and pj + 1 < N)
                for pmod in range(3):
                    if dp[pi][pj][pmod] == NEG_INF:
                        continue

                    is_forced = (move_num_mod == 0)

                        continue

                    cmod = (pmod + 1) % 3
                    if new_sum > dp[i][j][cmod]:
                        dp[i][j][cmod] = new_sum
                        prev[i][j][cmod] = (pi, pj, pmod, move)
    
    max_sum = NEG_INF
    best_mod = -1
    for m in range(3):
        if dp[N-1][N-1][m] > max_sum:

            best_mod = m
