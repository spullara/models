def solve_grid(grid: list[list[int]]) -> tuple[int, str]:
    n = len(grid)
    if n == 0:
        return (0, "")

    # best_sum/best_path[r][c][k]: max sum (all cells on the path, including
    # grid[r][c]) and the move string from (r, c) to the bottom-right corner,
    # given that k == (moves already taken) % 3.
    # Moves numbered 3, 6, 9, ... must be diagonal when that step stays in bounds,
    # which is exactly when k == 2 and X is in range.
    best_sum = [[[0] * 3 for _ in range(n)] for _ in range(n)]
    best_path = [[[""] * 3 for _ in range(n)] for _ in range(n)]

                continue

                can_x = r + 1 < n and c + 1 < n
                can_r = c + 1 < n
                can_d = r + 1 < n

                if must_x and can_x:
                    options = ("X",)
                else:
                    options = []
                    if can_r:
                        options.append("R")
                    if can_d:
                        options.append("D")
                    if can_x:
                        options.append("X")

                chosen_path = None
                nk = (k + 1) % 3
                for move in options:
                    if move == "R":
                        nr, nc = r, c + 1
                    elif move == "D":
                        nr, nc = r + 1, c
                    else:
                        nr, nc = r + 1, c + 1

                    path = move + best_path[nr][nc][nk]

                    if (
                        chosen_sum is None

                        chosen_path = path
